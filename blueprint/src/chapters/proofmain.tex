\begin{lemma}[Finiteness of the classes of tori]
\label{finite_gammaClass}
\lean{GSpinGGP.finite_gammaClass}
\leanok
\uses{GGPData, GammaClass}
$\Gamma(G,H)$ meets finitely many conjugacy classes of tori (Thesis \S6.5).
\end{lemma}
\begin{proof}
\leanok
\uses{LIT-finite_H1_normalizerBar}
Twisting argument (Serre, \emph{Cohomologie galoisienne} I \S5). Tori $S_y$ with the same $\dim W''_y$ are conjugate
under $\SO(W)(\bar F)$: $S_y(\bar F)$ is the split torus of a hyperbolic frame of eigenvectors of $P(y)$, and
Witt's extension theorem carries one frame to another of the same size; a conjugator $g$ gives the
cocycle $\sigma\mapsto g^{-1}\sigma(g)$ in $N_x=N_{\SO(W)}(S_x)$, and $H^1(F,N_x)$ is finite. Equal classes give a
Galois-fixed, hence rational, conjugator in $\SO(W)(F)$, which lifts to $\GSpin(W)(F)$ and conjugates the tori.
\end{proof}

\begin{lemma}[The torus isomorphism]
\label{exists_torusTransfer_core}
\lean{GSpinGGP.exists_torusTransfer_core}
\leanok
\uses{G, GGPData, GSpin, InnerFormSystem, InnerTwist, IsGammaElt, StablyConjG}
If $x\in\Gamma(G,H)$ and $y\in\Gamma(G_\alpha,H_\alpha)$ are strongly stably conjugate, then $\Ad(g)\circ\psi_\alpha$
restricts to an $F$-rational isomorphism of topological groups $T_x(F)\simeq T_y(F)$, the identity on scalars,
$x\mapsto y$, carrying $T_{x,\natural}$ onto $T_{y,\natural}$ with every $t$ stably conjugate to its image.
\end{lemma}
\begin{proof}
\leanok
\uses{isSemisimple_specP_baseChange, lieAbelian_specP_baseChange, stablyConjG_diag_of_psiU, steinberg_sign}
The values of $t\mapsto gC(\varphi_\alpha)(t)g^{-1}$ are Galois-fixed, hence rational
(Galois descent); they commute with $y$ and fix $W'_y$. The inverse comes from $g^{-1}$ and $\varphi_\alpha^{-1}$. Scalars are central and $x\mapsto y$ by the choice of $g$. $e$ and $e^{-1}$ are restrictions of $F$-linear maps $C(W)\to C(W_\alpha)$, hence continuous. Stable conjugacy in $G$ is Lemma~\ref{stablyConjG_diag_of_psiU}. On $T_{x,\natural}$ the centralizers of $\bar t$ and $\bar x$ in $\GSpin(W)(\Fbar)$ and $\GSpin(V)(\Fbar)$ agree (by the centralizer theorem, Theorem~\ref{steinberg_centralizer_connected}, and the commutativity of the Lie algebra of $S_t$); hence the rationality clauses for $x$ give those for $t$, and $\Ad(g)\circ C(\varphi_\alpha)$ carries centralizers, fixed spaces and semisimplicity, so $T_{x,\natural}$ and $T_{y,\natural}$ correspond.
\end{proof}

\begin{theorem}[Hilbert's Theorem 90 for $\mathrm{GL}_m$]
\label{hilbert90_GL}
\lean{GSpinGGP.hilbert90_GL}
\leanok

\leanok

\begin{theorem}[The Kottwitz sign is multiplicative on $H^1(F,S_x)$]
\label{kottwitzSign_twistIdx_mul}
\lean{GSpinGGP.kottwitzSign_twistIdx_mul}
\leanok
\uses{GGPData, GSpin, InnerFormSystem, InnerTwist, IsGammaElt, IsQuasiSplit, kottwitzSign}
For $G$ quasi-split and $x\in\Gamma(G,H)$, $\beta\mapsto e(G_{q_x(\beta)})$ is multiplicative on $H^1(F,S_x)$.
\end{theorem}
\begin{proof}
\leanok
\uses{kottwitzSign, padic_quadratic_classes, torusBarFbar_commute, nonempty_isometryEquiv_of_galObstruction_eq}
Own work, without Brauer groups (the statement is \cite{BP}, (12.1.6), with the sign of \cite{Kot83}).
$e(G_\alpha)=1$ exactly when $W_\alpha\cong W$. For a cocycle $a$ in $S_x$ choose lifts $t_\sigma\in T_x(\bar F)$;
the scalars $t_\sigma\,\sigma(t_\tau)\,t_{\sigma\tau}^{-1}$ form its obstruction. Call $a\sim b$ when they have lifts with
equal obstructions; as $T_x(\bar F)$ is commutative this is compatible with products. If $W_\alpha\cong W_\beta$, a lift
$h\in\mathrm{GSpin}(W)(\bar F)$ of $\varphi_a^{-1}\varphi_b$ (made proper by a rational reflection) conjugates a lift of $a$
into one of $b$, so $a\sim b$; conversely $a\sim b$ gives $W_\alpha\cong W_\beta$
(Lemma~\ref{nonempty_isometryEquiv_of_galObstruction_eq}). With at most two classes of the dimension and discriminant
of $W$, the sign is multiplicative.
\end{proof}

$\mathrm{GSpin}(U)(\bar F)$ with equal (scalar) obstructions, then $U_1\cong U_2$ over $F$.
\end{lemma}
\begin{proof}
\leanok
\uses{padic_quadratic_classes}
Own work. The quotient $v_\sigma=\varphi_1 s_\sigma t_\sigma^{-1}\varphi_1^{-1}$ of the lifts is a $1$-cocycle in
$\mathrm{GSpin}(U_1)(\bar F)$, and $C(\psi)$, $\psi=\varphi_2\varphi_1^{-1}$, satisfies $\sigma(C(\psi)z)=C(\psi)(v_\sigma\sigma(z)v_\sigma^{-1})$.
Hilbert 90 for the units of the central simple algebra $C(U_1)$ (even dimension) or $C^+(U_1)$ (odd dimension), deduced from
Hilbert 90 for $\mathrm{GL}_N$ and the fact that modules of equal dimension over a simple algebra are isomorphic, gives
$v_\sigma=u^{-1}\sigma(u)$; then $z\mapsto C(\psi)(u^{-1}zu)$ is Galois-equivariant and descends to $C(U_1)\cong C(U_2)$ (resp.
$C^+$). The discriminants agree since $\det\psi$ is Galois-fixed, so $U_1\cong U_2$ by the classification of quadratic forms over
$F$ by rank, discriminant and Clifford invariant.
\end{proof}

\leanok
